const ky = document.getElementById('xcdsasasasa').value;
const promptInput = document.getElementById('messageInput');

async function generateMarkdown(prompt) {
    const url = 'https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent';
    
    const requestBody = {
        contents: [{ parts: [{ text: prompt }] }],
        generationConfig: {
            temperature: 0.7,
            topK: 40,
            topP: 0.95,
            maxOutputTokens: 1024,
        },
    };

    try {
        const response = await fetch(`${url}?key=${ky}`, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify(requestBody),
        });

        if (!response.ok) {
            throw new Error('API request failed');
        }

        const data = await response.json();
        return data.candidates[0].content.parts[0].text;
    } catch (error) {
        console.error('Error:', error);
        return 'An error occurred while generating the markdown.';
    }
}

// Event listener for the generate button
generateButton.addEventListener('click', async () => {
    if (promptCount < 5) { // Check if prompt limit is not reached
        const prompt = promptInput.value.trim();
        if (prompt) {
            editor.setValue('Generating...');
            let markdown;
            
            // Check for custom responses before calling Gemini API
            const customResponse = handleCustomResponse(prompt);
            if (customResponse) {
                editor.setValue(customResponse);
                promptCount++; // Increment prompt counter
            } else {
                markdown = await generateMarkdown(prompt);
                editor.setValue(markdown);
                promptCount++; // Increment prompt counter
            }
            
            if (promptCount === 5) {
                generateButton.disabled = true; // Disable button when limit is reached
                alert('You have reached the maximum limit of prompts (5).');
            }
        } else {
            alert('Please enter a prompt.');
        }
    } else {
        alert('You have reached the maximum limit of prompts (5).');
    }
});
